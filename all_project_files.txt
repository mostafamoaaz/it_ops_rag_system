from typing import Any
from documents import DOCUMENTS


def chunk_document(
    doc: dict[str, Any],
    chunk_size: int = 40,
    overlap: int = 10
) -> list[dict[str, Any]]:
    """
    Splits a single document's text into fixed-size chunks with overlap.
    Inherits parent metadata to enable downstream metadata filtering and source citation.
    """
    text = doc.get("text", "").strip()
    words = text.split()
    
    if not words:
        return []

    # If the document is shorter than the chunk size, keep it intact
    if len(words) <= chunk_size:
        return [
            {
                "chunk_id": f"{doc['document_id']}_c0",
                "document_id": doc["document_id"],
                "chunk_index": 0,
                "title": doc.get("title", ""),
                "department": doc.get("department", ""),
                "doc_type": doc.get("doc_type", ""),
                "effective_date": doc.get("effective_date", ""),
                "is_current": doc.get("is_current", True),
                "text": text,
            }
        ]

    chunks = []
    step = max(1, chunk_size - overlap)
    chunk_idx = 0

    for i in range(0, len(words), step):
        chunk_words = words[i : i + chunk_size]
        chunk_text = " ".join(chunk_words)

        chunks.append(
            {
                "chunk_id": f"{doc['document_id']}_c{chunk_idx}",
                "document_id": doc["document_id"],
                "chunk_index": chunk_idx,
                "title": doc.get("title", ""),
                "department": doc.get("department", ""),
                "doc_type": doc.get("doc_type", ""),
                "effective_date": doc.get("effective_date", ""),
                "is_current": doc.get("is_current", True),
                "text": chunk_text,
            }
        )
        chunk_idx += 1

        # Stop when the remaining words are covered
        if i + chunk_size >= len(words):
            break

    return chunks


def build_chunked_corpus(
    documents: list[dict[str, Any]],
    chunk_size: int = 40,
    overlap: int = 10
) -> list[dict[str, Any]]:
    """
    Generates a flattened list of all chunks across the document corpus.
    """
    all_chunks = []
    for doc in documents:
        doc_chunks = chunk_document(doc, chunk_size=chunk_size, overlap=overlap)
        all_chunks.extend(doc_chunks)
    return all_chunks

"""
context_building.py - Context Package Builder
Formats retrieved chunks, enforces strict token/word budgets, and deduplicates content.
"""

from typing import Any
from retrieve import hybrid_search

# Maximum words allowed in the prompt's context block
DEFAULT_WORD_BUDGET = 250


def build_context_package(
    chunks: list[dict[str, Any]],
    word_budget: int = DEFAULT_WORD_BUDGET,
    include_metadata: bool = True
) -> str:
    """
    Formats retrieved chunks into a structured context string for prompt injection.
    Stops adding chunks when the word budget is reached.
    """
    if not chunks:
        return "No relevant operational runbooks found."

    package_blocks = []
    accumulated_words = 0
    seen_chunk_ids = set()

    for chunk in chunks:
        if chunk.get("hybrid_score", 0) < 0.20:
            continue
        c_id = chunk["chunk_id"]
        if c_id in seen_chunk_ids:
            continue
        seen_chunk_ids.add(c_id)

        meta = chunk.get("metadata", {})
        title = meta.get("title", "Unknown Procedure")
        dept = meta.get("department", "Operations")
        doc_type = meta.get("doc_type", "runbook")
        text = chunk.get("text", "").strip()

        if include_metadata:
            block = f"[{title} | Dept: {dept} | Type: {doc_type}]\n{text}"
        else:
            block = text

        block_word_count = len(block.split())

        # If adding this block exceeds the budget, check if we already have some context
        if accumulated_words + block_word_count > word_budget:
            if not package_blocks:
                # If even the first chunk exceeds, truncate it to fit
                words = block.split()[:word_budget]
                package_blocks.append(" ".join(words) + " ...")
            break

        package_blocks.append(block)
        accumulated_words += block_word_count

    return "\n\n---\n\n".join(package_blocks)


def retrieve_and_build_context(
    query: str,
    top_k: int = 5,
    alpha: float = 0.5,
    word_budget: int = DEFAULT_WORD_BUDGET,
    current_only: bool = True
) -> tuple[str, list[dict[str, Any]]]:
    """
    End-to-end convenience function: retrieves candidate chunks and packages them.
    Returns (context_text, raw_chunks).
    """
    chunks = hybrid_search(query=query, top_k=top_k, alpha=alpha, current_only=current_only)
    context = build_context_package(chunks, word_budget=word_budget)
    return context, chunks


if __name__ == "__main__":

from pathlib import Path
import chromadb

from documents import DOCUMENTS
from chunking import build_chunked_corpus
from vector_representation import get_dense_embeddings

# Define persistent database path
CHROMA_PATH = Path("./chroma_db")
COLLECTION_NAME = "it_ops_runbooks"


def initialize_vector_store():
    """
    Creates or connects to a persistent ChromaDB instance and populates it with chunk embeddings.
    """
    print(f"Connecting to persistent ChromaDB at: {CHROMA_PATH.resolve()}...")
    client = chromadb.PersistentClient(path=str(CHROMA_PATH))

    # Retrieve or create collection using cosine similarity space
    collection = client.get_or_create_collection(
        name=COLLECTION_NAME,
        metadata={"hnsw:space": "cosine"}
    )

    print("Generating chunks from operational documents...")
    chunks = build_chunked_corpus(DOCUMENTS, chunk_size=40, overlap=10)
    print(f"Total chunks to index: {len(chunks)}")

    ids = [c["chunk_id"] for c in chunks]
    texts = [c["text"] for c in chunks]

    # ChromaDB metadata values must be int, float, str, or bool
    metadatas = [
        {
            "document_id": c["document_id"],
            "chunk_index": c["chunk_index"],
            "title": c["title"],
            "department": c["department"],
            "doc_type": c["doc_type"],
            "effective_date": c["effective_date"],
            "is_current": c["is_current"],
        }
        for c in chunks
    ]

    print("Encoding dense embeddings for chunks...")
    embeddings = get_dense_embeddings(texts).tolist()

    print("Upserting records into ChromaDB collection...")
    collection.upsert(
        ids=ids,
        embeddings=embeddings,
        documents=texts,
        metadatas=metadatas
    )

    count = collection.count()
    print(f"âœ… Successfully initialized '{COLLECTION_NAME}'. Total indexed chunks in ChromaDB: {count}")
    return client, collection


if __name__ == "__main__":
    initialize_vector_store()

import os

from dotenv import load_dotenv
from google import genai

load_dotenv()


@lru_cache(maxsize=1)
def get_best_available_model(client: genai.Client = None) -> str:
    """Discovers active models for the API key and returns the best available Flash model."""
    if client is None:
        api_key = os.getenv("GEMINI_API_KEY")
        if not api_key:
            raise ValueError("GEMINI_API_KEY not found in environment!")
        client = genai.Client(api_key=api_key)

    # Preferred models in priority order
    priority_order = [
        "gemini-3.5-flash-lite",
        "gemini-3.5-flash",
        "gemini-flash-lite-latest",
        "gemini-flash-latest",
        "gemini-3.8-flash",
    ]

    try:
        available_names = [m.name.replace("models/", "") for m in client.models.list()]

        for pref in priority_order:
            if pref in available_names:
                return pref

        # Fallback to the first model matching 'flash'
        for name in available_names:
            if "flash" in name:
                return name

        return available_names[0] if available_names else "gemini-3.5-flash-lite"
    except Exception as e:
        print(f"âš ï¸ Model discovery failed ({e}), falling back to gemini-3.5-flash-lite")
        return "gemini-3.5-flash-lite"


if __name__ == "__main__":

    selected = get_best_available_model()

def tokenize(text: str) -> list[str]:
    """
    Extracts lowercase alphanumeric tokens, preserving technical identifiers like
    port numbers, error codes, and server names.
    """
    return re.findall(r"[a-z0-9_\-\.]+", text.lower())



def preprocess_text(text: str, remove_stopwords: bool = True) -> list[str]:
    """
    Tokenizes text and selectively removes stopwords while preserving critical
    operational negation and status keywords.
    """
    tokens = tokenize(text)
    if remove_stopwords:
        tokens = [t for t in tokens if t not in STOPWORDS]
    return tokens


def build_search_text(doc: dict) -> str:
    """Enriches operational document text with parent metadata for hybrid indexing."""
    parts = [
        f"Title: {doc.get('title', '')}",
        f"Department: {doc.get('department', '')}",
        f"Type: {doc.get('doc_type', '')}",
        doc.get("text", "")
    ]
    return "\n".join(parts)

def generate_rag_response(
    query: str,
    alpha: float = 0.5,
    top_k: int = 4,
    word_budget: int = 350,
    model: str = None
) -> dict:
    """
    Executes hybrid retrieval, builds context, and invokes the selected Gemini model.
    Automatically resolves the best active model if none is explicitly provided.
    """
    # Dynamically select active model from list_gemini_models.py if unspecified
    if not model:
        model = get_best_available_model(client)

    # Retrieve candidate chunks and package within word budget
    context, retrieved_chunks = retrieve_and_build_context(
        query=query,
        top_k=top_k,
        alpha=alpha,
        word_budget=word_budget,
        current_only=True
    )

    user_message = format_user_prompt(query, context)
    answer = None

    # Retry loop with exponential backoff for handling transient 503 / rate spike errors
    for attempt in range(3):
        try:
            response = client.models.generate_content(
                model=model,
                contents=user_message,
                config=types.GenerateContentConfig(
                    system_instruction=SYSTEM_PROMPT,
                    temperature=0.0,  # Deterministic output for technical procedures
                ),
            )
            answer = response.text
            break
        except Exception as e:
            err_msg = str(e)
            if ("503" in err_msg or "UNAVAILABLE" in err_msg) and attempt < 2:
                time.sleep(2 * (attempt + 1))
                continue
            answer = f"âš ï¸ Gemini API Error on model '{model}': {err_msg}\n\n(Retrieved context is still accessible below)."
            break

    return {
        "query": query,
        "answer": answer,
        "context": context,
        "chunks": retrieved_chunks,
        "model_used": model
    }

def get_chroma_collection():
    """
    Connects to the persistent ChromaDB collection.
    If running on Streamlit Cloud and chroma_db doesn't exist yet,
    initializes and populates the store automatically.
    """
    if not CHROMA_PATH.exists() or not any(CHROMA_PATH.iterdir()):
        print("ChromaDB storage not found. Building index from documents...")
        client, collection = initialize_vector_store()
        return collection

    client = chromadb.PersistentClient(path=str(CHROMA_PATH))
    return client.get_collection(name=COLLECTION_NAME)


def hybrid_search(
    query: str,
    top_k: int = 5,
    alpha: float = 0.5,
    current_only: bool = True
) -> list[dict[str, Any]]:
    """
    Executes hybrid retrieval over the corpus.
    
    Args:
        query: User prompt or technical incident question.
        top_k: Number of candidate chunks to return.
        alpha: Weight for dense vs lexical (1.0 = pure semantic, 0.0 = pure BM25).
        current_only: If True, filters out superseded/outdated operational runbooks.
    """
    collection = get_chroma_collection()  # Fixed here!

    # Apply metadata filter if current_only is True
    where_filter = {"is_current": True} if current_only else None

    # Retrieve all eligible records from ChromaDB
    all_records = collection.get(
        where=where_filter,
        include=["documents", "metadatas", "embeddings"]
    )

    doc_texts = all_records["documents"]
    metadatas = all_records["metadatas"]
    chunk_ids = all_records["ids"]
    stored_embeddings = np.array(all_records["embeddings"])

    if not doc_texts:
        return []

    # 1. Compute Dense Semantic Similarity
    query_embed = get_dense_embeddings([query])
    # Dot product with L2-normalized embeddings equals cosine similarity
    dense_scores = np.dot(stored_embeddings, query_embed.T).flatten()

    # 2. Compute BM25 Lexical Scores
    bm25_corpus = [{"text": t} for t in doc_texts]
    bm25 = build_bm25_index(bm25_corpus)
    q_tokens = preprocess_text(query)
    bm25_scores = np.array(bm25.get_scores(q_tokens))

    # 3. Fuse via Normalized Hybrid Formula
    final_scores = compute_hybrid_scores(dense_scores, bm25_scores, alpha=alpha)

    # 4. Rank and select top_k candidates
    top_indices = np.argsort(final_scores)[::-1][:top_k]

    results = []
    for idx in top_indices:
        results.append({
            "chunk_id": chunk_ids[idx],
            "text": doc_texts[idx],
            "metadata": metadatas[idx],
            "dense_score": float(dense_scores[idx]),
            "bm25_score": float(bm25_scores[idx]),
            "hybrid_score": float(final_scores[idx]),
        })

    return results

from typing import Any
import numpy as np
from rank_bm25 import BM25Okapi
from preprocessing import preprocess_text

MODEL_NAME = "all-MiniLM-L6-v2"
_model = None


def get_embedding_model():
    """Lazy-loads the SentenceTransformer model on first actual usage."""
    global _model
    if _model is None:
        print(f"Loading dense embedding model: {MODEL_NAME}...")
        from sentence_transformers import SentenceTransformer
        _model = SentenceTransformer(MODEL_NAME)
    return _model


def get_dense_embeddings(texts: list[str]) -> np.ndarray:
    model = get_embedding_model()
    return model.encode(
        texts,
        normalize_embeddings=True,
        convert_to_numpy=True,
        show_progress_bar=False,
    )


def build_bm25_index(chunks: list[dict[str, Any]]) -> BM25Okapi:
    tokenized_corpus = [preprocess_text(c["text"]) for c in chunks]
    return BM25Okapi(tokenized_corpus)


def min_max_normalize(scores: np.ndarray) -> np.ndarray:
    min_val = np.min(scores)
    max_val = np.max(scores)
    if max_val == min_val:
        return np.zeros_like(scores, dtype=float)
    return (scores - min_val) / (max_val - min_val)


def compute_hybrid_scores(
    dense_scores: np.ndarray, lexical_scores: np.ndarray, alpha: float = 0.5
) -> np.ndarray:
    norm_dense = min_max_normalize(dense_scores)
    norm_lexical = min_max_normalize(lexical_scores)
    return alpha * norm_dense + (1.0 - alpha) * norm_lexical
